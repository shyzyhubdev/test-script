local function classify(obj)
    -- GIỮ ANIMATION
    if obj:IsA("Animator")
        or obj:IsA("Animation")
        or obj:IsA("AnimationController") then
        return false
    end

    -- BLOCK / PART / MESH / UNION
    if obj:IsA("BasePart") then
        if obj:IsA("Terrain") then
            return false
        end
        if obj:IsA("MeshPart") then
            return "MeshPart"
        end
        if obj:IsA("PartOperation") then
            return "Union"
        end
        return "Part"
    end

    -- TEXTURE
    if REMOVE_TEXTURE then
        if obj:IsA("Decal")
            or obj:IsA("SurfaceAppearance")
            or obj:IsA("MaterialVariant")
            or obj:IsA("Clothing")
            or obj:IsA("ShirtGraphic") then
            return "Destroy"
        end

        if obj:IsA("SpecialMesh") then
            return "SpecialMesh"
        end
    end

    -- EFFECT (Destroy thay vì Enabled=false: game gọi :Emit() hoặc bật lại Enabled vẫn không có tác dụng)
    if REMOVE_EFFECTS then
        if obj:IsA("ParticleEmitter")
            or obj:IsA("Trail")
            or obj:IsA("Beam")
            or obj:IsA("Smoke")
            or obj:IsA("Fire")
            or obj:IsA("Sparkles")
            or obj:IsA("Highlight")
            or obj:IsA("ForceField") then
            return "Destroy"
        end
    end

    -- EXPLOSION
    if REMOVE_EXPLOSIONS and obj:IsA("Explosion") then
        return "Explosion"
    end

    -- LIGHT
    if REMOVE_LIGHTS and obj:IsA("Light") then
        return "Destroy"
    end

    -- POST EFFECT / ATMOSPHERE / CLOUDS
    if REMOVE_POST_EFFECTS then
        if obj:IsA("PostEffect") then
            return "PostEffect"
        end
        if obj:IsA("Atmosphere") then
            return "Atmosphere"
        end
        if obj:IsA("Clouds") then
            return "Clouds"
        end
    end

    -- SOUND
    if MUTE_SOUNDS and obj:IsA("Sound") then
        return "Mute"
    end

    if REMOVE_SOUND_FX and obj:IsA("SoundEffect") then
        return "SoundFx"
    end

    -- PHỤ KIỆN NGƯỜI CHƠI KHÁC
    if SIMPLIFY_OTHER_PLAYERS and obj:IsA("Accoutrement") then
        return "Accessory"
    end

    -- BILLBOARD
    if HIDE_BILLBOARDS and obj:IsA("BillboardGui") then
        return "Billboard"
    end

    return false
end

local kindCache = {}

local function kindOf(obj)
    local className = obj.ClassName
    local kind = kindCache[className]

    if kind == nil then
        kind = classify(obj)
        kindCache[className] = kind
    end

    return kind
end

local darkened = setmetatable({}, { __mode = "k" })      -- part đã tối rồi: không tối thêm khi bị gắn lại vào workspace
local postHooked = setmetatable({}, { __mode = "k" })

local function darkColor(color)
    return Color3.new(
        color.R * DARK_AMOUNT,
        color.G * DARK_AMOUNT,
        color.B * DARK_AMOUNT
    )
end

local function hPart(p)
    if not darkened[p] then
        darkened[p] = true
        p.Color = darkColor(p.Color)
    end

    p.Material = SMOOTH
    p.Reflectance = 0

    if DISABLE_SHADOWS then
        p.CastShadow = false
    end

    if REMOVE_TEXTURE then
        p.MaterialVariant = ""
    end
end

local function meshExtras(p)
    p.RenderFidelity = PERFORMANCE

    if REMOVE_TEXTURE then
        p.TextureID = ""
    end
end

local function hMesh(p)
    pcall(hPart, p)
    pcall(meshExtras, p)
end

local function unionExtras(p)
    p.UsePartColor = true
    p.RenderFidelity = PERFORMANCE

    if REMOVE_TEXTURE then
        p.TextureID = ""
    end
end

local function hUnion(p)
    pcall(hPart, p)
    pcall(unionExtras, p)
end

local function hDestroy(o)
    o:Destroy()
end

local function hSpecialMesh(o)
    o.TextureId = ""
end

local function hExplosion(o)
    o.Visible = false
    o.BlastPressure = 0
end

local function hPost(o)
    o.Enabled = false

    -- game bật lại thì tắt lại (mỗi effect chỉ gắn 1 lần)
    if not postHooked[o] then
        postHooked[o] = true
        o:GetPropertyChangedSignal("Enabled"):Connect(function()
            if running and o.Enabled then
                o.Enabled = false
            end
        end)
    end
end

local function hAtmosphere(o)
    o.Density = 0
    o.Haze = 0
    o.Glare = 0
end

local function hClouds(o)
    o.Enabled = false
end

local function hMute(o)
    o.Volume = 0
end

local function hSoundFx(o)
    o.Enabled = false
end

local function hAccessory(o)
    local model = o.Parent
    if not model or not model:IsA("Model") then
        return
    end

    -- nhân vật của người chơi khác (hoặc clone/tàn ảnh mang tên họ); của mình thì giữ nguyên
    local plr = Players:GetPlayerFromCharacter(model) or Players:FindFirstChild(model.Name)

    if plr and plr:IsA("Player") and plr ~= Player then
        o:Destroy()
    end
end

local function hBillboard(o)
    o.Enabled = false
end

local fxEnabled = HIDE_FX_PARTS
local fxQueue = {}                                  -- { {part, hạn hiện lại, hạn kiểm tra chủ nhân} }
local fxHidden = setmetatable({}, { __mode = "k" })

-- Tên các bộ phận cơ thể thật (R6 + R15) — part khác gắn trên nhân vật được coi là hiệu ứng
local BODY_PARTS = {
    ["Head"] = true, ["Torso"] = true,
    ["Left Arm"] = true, ["Right Arm"] = true,
    ["Left Leg"] = true, ["Right Leg"] = true,
    ["HumanoidRootPart"] = true,
    ["UpperTorso"] = true, ["LowerTorso"] = true,
    ["LeftUpperArm"] = true, ["LeftLowerArm"] = true, ["LeftHand"] = true,
    ["RightUpperArm"] = true, ["RightLowerArm"] = true, ["RightHand"] = true,
    ["LeftUpperLeg"] = true, ["LeftLowerLeg"] = true, ["LeftFoot"] = true,
    ["RightUpperLeg"] = true, ["RightLowerLeg"] = true, ["RightFoot"] = true,
}

-- Tìm nhân vật THẬT của người chơi chứa part này (clone tàn ảnh không tính vì không thuộc Player nào)
local function ownerCharacter(part)
    local model = part:FindFirstAncestorOfClass("Model")

    while model do
        if Players:GetPlayerFromCharacter(model) then
            return model
        end
        model = model:FindFirstAncestorOfClass("Model")
    end

    return nil
end

-- true = part này là hiệu ứng/tàn ảnh -> ẩn
local function shouldHide(part)
    local char = ownerCharacter(part)

    -- không thuộc nhân vật người chơi nào: hiệu ứng, tàn ảnh (clone), part tạm...
    if not char then
        return true
    end

    -- phụ kiện, vũ khí của nhân vật thật: giữ
    if part:FindFirstAncestorWhichIsA("Accoutrement") or part:FindFirstAncestorOfClass("Tool") then
        return false
    end

    -- bộ phận cơ thể thật: giữ
    if part.Parent == char and BODY_PARTS[part.Name] then
        return false
    end

    -- part lạ gắn trên người (nước, aura dạng mesh...): ẩn
    return true
end

local function setHidden(part)
    part.LocalTransparencyModifier = 1
end

local function setShown(part)
    part.LocalTransparencyModifier = 0
end

local function showFx(part)
    if fxHidden[part] then
        fxHidden[part] = nil
        pcall(setShown, part)

        -- part này chưa được tô tối/đổi material lúc đang ẩn: đưa vào hàng đợi xử lý
        if part.Parent then
            enqueueNew(part)
        end
    end
end

-- trả về true nếu part đã bị ẩn (khi đó bỏ qua tô tối/đổi material cho đỡ tốn việc)
local function hideFx(obj)
    if not fxEnabled then
        return false
    end

    -- đang ẩn quá nhiều part cùng lúc = nhiều khả năng là map vừa spawn, không phải hiệu ứng
    if #fxQueue >= FX_MAX_HIDDEN then
        return false
    end

    -- game bật StreamingEnabled: map liên tục spawn part -> bỏ qua part tĩnh (anchored + CanCollide)
    -- để map không bị "tàng hình" 4 giây mỗi lần stream
    if STREAMING and obj.Anchored and obj.CanCollide then
        return false
    end

    if not shouldHide(obj) then
        return false
    end

    pcall(setHidden, obj)

    local now = os.clock()
    fxHidden[obj] = true
    fxQueue[#fxQueue + 1] = { obj, now + FX_KEEP_SECONDS, now + RECHECK_SECONDS }

    return true
end

local function setFxEnabled(state)
    fxEnabled = state

    if not fxEnabled then
        for _, item in ipairs(fxQueue) do
            showFx(item[1])
        end
        table.clear(fxQueue)
    end
end

-- Dọn hàng đợi: part biến mất -> bỏ; sống quá lâu hoặc hoá ra là nhân vật -> hiện lại
-- (chỉ kiểm tra "có phải nhân vật không" trong RECHECK_SECONDS đầu, sau đó chỉ xét hạn -> nhẹ hơn nhiều khi đông)
task.spawn(function()
    while running do
        task.wait(0.25)

        local now = os.clock()
        local i = 1

        while i <= #fxQueue do
            local item = fxQueue[i]
            local part = item[1]

            local remove = false

            if not part.Parent then
                remove = true
            elseif now >= item[2] or (now < item[3] and not shouldHide(part)) then
                showFx(part)
                remove = true
            end

            if remove then
                fxQueue[i] = fxQueue[#fxQueue]
                fxQueue[#fxQueue] = nil
            else
                i += 1
            end
        end
    end
end)

    if gameProcessed or input.KeyCode ~= TOGGLE_KEY then
        return
    end

    setFxEnabled(not fxEnabled)
end))

local farChars = setmetatable({}, { __mode = "k" })      -- [character] = { các part đang bị ẩn }

local function setCharFar(char, hide)
    if hide then
        local parts = {}

        for _, d in ipairs(char:GetDescendants()) do
            if d:IsA("BasePart") then
                parts[#parts + 1] = d
                pcall(setHidden, d)
            end
        end

        farChars[char] = parts
    else
        local parts = farChars[char]
        farChars[char] = nil

        if parts then
            for _, p in ipairs(parts) do
                pcall(setShown, p)
            end
        end
    end
end

local function updateFar()
    local active = FAR_PLAYER_DISTANCE > 0 and (crowd or fpsAvg < LOW_FPS)

    local camera = workspace.CurrentCamera
    if not camera then
        return
    end

    local origin = camera.Focus.Position

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= Player then
            local char = plr.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")

            if root then
                local isHidden = farChars[char] ~= nil
                local dist = (root.Position - origin).Magnitude

                if not isHidden then
                    if active and dist > farRadius then
                        setCharFar(char, true)
                    end
                elseif (not active) or dist < farRadius * 0.85 then
                    setCharFar(char, false)
                end
            end
        end
    end
end

-- mỗi giây: đếm người + thu/nới khoảng cách theo FPS
task.spawn(function()
    while running do
        task.wait(1)

        pcall(function()
            crowd = #Players:GetPlayers() >= CROWD_PLAYERS

            if fpsAvg < LOW_FPS then
                farRadius = math.max(FAR_PLAYER_MIN, farRadius - 15)
            elseif fpsAvg > GOOD_FPS then
                farRadius = math.min(FAR_PLAYER_DISTANCE, farRadius + 10)
            end
        end)
    end
end)

-- mỗi 0.5 giây: ẩn/hiện người chơi theo khoảng cách
task.spawn(function()
    while running do
        task.wait(0.5)
        pcall(updateFar)
    end
end)

-- XỬ LÝ 1 OBJECT + BƠM HÀNG ĐỢI THEO NGÂN SÁCH THỜI GIAN
--------------------------------------------------

local function process(obj)
    local kind = kindOf(obj)
    if not kind then
        return
    end

    -- đã bị xoá trước khi tới lượt
    if not obj.Parent then
        return
    end

    local handler = HANDLERS[kind]
    if handler then
        pcall(handler, obj)
    end
end

local function pump(budget)
    local t0 = os.clock()
    local n = 0

    -- 1) part mới spawn trước
    while nqHead <= nqTail do
        local obj = newQueue[nqHead]
        newQueue[nqHead] = nil
        nqHead += 1

        process(obj)

        n += 1
        if n % 8 == 0 and os.clock() - t0 >= budget then
            return
        end
    end

    nqHead, nqTail = 1, 0

    -- 2) quét map có sẵn theo từng mảnh nhỏ (không GetDescendants cả map một lúc -> không đứng hình)
    while true do
        if not curList then
            if nodeHead > nodeTail then
                return
            end

            local inst = nodeInst[nodeHead]
            local depth = nodeDepth[nodeHead]
            nodeInst[nodeHead] = nil
            nodeDepth[nodeHead] = nil
            nodeHead += 1

            if inst.Parent then
                process(inst)

                if depth < MAX_SPLIT_DEPTH and (inst:IsA("Folder") or inst:IsA("Model")) then
                    local ok, children = pcall(inst.GetChildren, inst)
                    if ok and children then
                        for i = 1, #children do
                            addNode(children[i], depth + 1)
                        end
                    end
                else
                    local ok, list = pcall(inst.GetDescendants, inst)
                    if ok and list and #list > 0 then
                        curList, curTotal, curIdx = list, #list, 1
                    end
                end
            end

            n += 1
            if os.clock() - t0 >= budget then
                return
            end
        else
            while curIdx <= curTotal do
                local obj = curList[curIdx]
                curIdx += 1

                process(obj)

                n += 1
                if n % 16 == 0 and os.clock() - t0 >= budget then
                    return
                end
            end

            curList = nil
        end
    end
end

local fpsFrames = 0
local fpsElapsed = 0

track(RunService.RenderStepped:Connect(function(dt)
    -- đo FPS (dùng cho dòng FPS và chế độ đông người)
    fpsFrames += 1
    fpsElapsed += dt

    if fpsElapsed >= 0.5 then
        local fps = fpsFrames / fpsElapsed
        fpsAvg = fpsAvg * 0.5 + fps * 0.5
        fpsFrames = 0
        fpsElapsed = 0

        if onFpsSample then
            pcall(onFpsSample, fps)
        end
    end

    -- dọn dẹp: ngân sách tự tăng khi FPS thấp (v3 cố định nên FPS tụt là dọn càng chậm -> tồn việc)
    local scanning = curList ~= nil or nodeHead <= nodeTail

    if scanning or nqHead <= nqTail then
        local heavy = scanning or (nqTail - nqHead) > 1500
        local base = heavy and LOAD_BUDGET or LIVE_BUDGET

local function onAdded(obj)
    local kind = kindOf(obj)
    if not kind then
        return
    end

    if PART_KINDS[kind] then
        -- hiệu ứng dạng part: ẩn NGAY (rẻ), không tốn công tô tối nếu nó chỉ sống vài giây
        if fxEnabled then
            local ok, hidden = pcall(hideFx, obj)
            if ok and hidden then
                return
            end
        end

        enqueueNew(obj)
        return
    end

    -- particle/beam/trail/light/texture...: xoá NGAY, không xếp hàng nên không bị trễ khi đông
    local handler = HANDLERS[kind]
    if handler then
        pcall(handler, obj)
    end
end

            end
        end
    end
end)

-- ANTI SCREEN SHAKE (nhẹ hơn: cache Humanoid, chỉ ghi camera khi thật sự bị nghiêng)

local cachedHum = nil

local function antiShakeStep()
    local camera = workspace.CurrentCamera
    if not camera then
        return
    end

    -- bỏ rung do CameraOffset
    local hum = cachedHum
    if not hum or not hum.Parent then
        local char = Player.Character
        hum = char and char:FindFirstChildOfClass("Humanoid")
        cachedHum = hum
    end

    if hum and hum.CameraOffset ~= Vector3.zero then
        hum.CameraOffset = Vector3.zero
    end

    -- bỏ roll/nghiêng: camera không nghiêng thì RightVector nằm ngang (Y = 0) -> khỏi ghi lại CFrame
    local cf = camera.CFrame

    if math.abs(cf.RightVector.Y) > 0.0005 then
        local look = cf.LookVector

        if math.abs(look.Y) < 0.999 then
            local pos = cf.Position
            camera.CFrame = CFrame.lookAt(pos, pos + look, Vector3.yAxis)
        end
    end
end

if ANTI_SCREEN_SHAKE then
    pcall(function()
        RunService:UnbindFromRenderStep("AntiScreenShake")
    end)

    RunService:BindToRenderStep(
        "AntiScreenShake",
        Enum.RenderPriority.Camera.Value + 1,
        function()
            pcall(antiShakeStep)
        end
    )
end

local ramFailed = false

local function getRam()
    if ramFailed then
        return nil
    end

    local ok, mb = pcall(function()
        return Stats:GetTotalMemoryUsageMb()
    end)

    if ok and type(mb) == "number" then
        return math.floor(mb)
    end

    ramFailed = true
    return nil
end
